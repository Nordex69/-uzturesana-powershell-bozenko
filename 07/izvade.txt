PS C:\Users\bruno> $disks = Get-CimInstance -ClassName Win32_LogicalDisk -Filter "DriveType = 3"
PS C:\Users\bruno> $threshold = 50.0
PS C:\Users\bruno> foreach ($disk in $disks) {
    $letter = $disk.DeviceID
    $totalGb = [Math]::Round($disk.Size / 1GB, 2)
    $freeGb = [Math]::Round($disk.FreeSpace / 1GB, 2)
    
    Write-Output "Disk $letter | Total size: $totalGb GB | Free space: $freeGb GB"
    
    if ($freeGb -lt $threshold) {
        Write-Warning "Warning! Disk $letter free space ($freeGb GB) is below the threshold ($threshold GB)."
    }
}
